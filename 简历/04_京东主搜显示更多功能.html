    京东主搜显示更多功能

    李征2025/8/8大约 3 分钟

    功能介绍

    显示更多效果

    点击更多，显示收起效果

    产品要求：

    1. 筛选按钮展示不下换行，显示“更多”按钮，默认不显示
    2. 更多收起按钮始终显示在第一行最后一个按钮后面24px，而不是固定居右

    实现思路

    大致思路如下

    以下都是伪代码展示

    1. 收集所有筛选按钮宽度
    // 加载信息
const boxLoadedMap = new Map<StringOrNumber, number>()

// 计算筛选元素占据宽度总和
const computedTotalSize = () => {
  if (boxLoadedMap.size > 0 && boxLoadedMap.size === _filterItemList.length) {
    // 筛选项宽度总和
    const sumWidth: number = [...boxLoadedMap.values()].reduce((a, b) => a + b, 0)
    // 每项间隔12px
    const sumSpace: number = (boxLoadedMap.size - 1) * 12
    // 触发 onSizeChange 事件
    onSizeChange(sumWidth + sumSpace, boxLoadedMap)
  }
}

const handleBoxLoaded = (width: number, item: CheckBoxItemType) => {
    boxLoadedMap.set(item._id, width)
    // boxLoadedMap.size 等于筛选按钮列表的长度时表示全部加载完成
    if (boxLoadedMap.size === _filterItemList.length) {
        // 触发计算
        computedTotalSize()
    }
}

<FilterSelect onLoaded={(width) => handleBoxLoaded(width, item)} />

    1. 计算更多按钮偏移量

    第 3 步代码改造如下

    let boxLoadedWidths: number[] = []

// 计算更多按钮偏移量
const computeWidth = (useWidth: number) => {
  let sum = 0
  for (let i = 0, len = boxLoadedWidths.length; i < len; i++) {
    const newSum = sum + boxLoadedWidths[i] + 12 // 每一项间隔12px
    if (newSum > useWidth) {
      // 证明换行了 但是显示“更多”按钮后有可能会将第一行的末尾元素挤到第二行
      // 重置可用宽度
      const _useWidth = useWidth - moreWidth
      if (sum >= _useWidth) {
        // 挤下去了最后一个
        return _useWidth - (sum - boxLoadedWidths[i - 1])
      } else {
        // 没有挤下去
        return _useWidth - sum
      }
    } else {
      sum = newSum
    }
  }
  return sum
}

useEffect(() => {

const handleSizeChange = (width: number, boxLoadedMap: Map<StringOrNumber, number>) => {
  boxLoadedWidths = new Array(boxLoadedMap.size)
  for (const [k, v] of boxLoadedMap.entries()) {
    // 从左到右更具索引将每个筛选按钮的尺寸放到数组中
    const i = parseInt((k as string).split('-').pop()!)
    boxLoadedWidths[i] = v
  }
  setSumWidth(width)
}
