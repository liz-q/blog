  // 3. hash 字段：如果参数中有 # 会被认为是 hash，此时可以截取 ? 后面的字符串，用 URLSearchParams 解析
  const {search, hash, searchParams} = urlObj

  let _searchParams = searchParams

  if (hash) {
    const index = url.indexOf('?')
    const searchStr = url.substring(index)

  }

  const queryParams = {}

    queryParams[k] = v
  }
  return queryParams

// 解析函数
function getQuery(url) {
  // 创建一个 url 对象
  const urlObj = new URL(url)
  // console.log(urlObj)

  // 1. search 字段：?后面的字符串，包含?
  // 2. searchParams 字段：URLSearchParams 对象，search 中的各个查询参数。等价于 new URLSearchParams(search)
  // 3. hash 字段：如果参数中有 # 会被认为是 hash，此时可以截取 ? 后面的字符串，用 URLSearchParams 解析
  const {search, hash, searchParams} = urlObj

  let _searchParams = searchParams

  if (hash) {
    const index = url.indexOf('?')
    const searchStr = url.substring(index)
    _searchParams = new URLSearchParams(searchStr)
  }

  const queryParams = {}
  for (const [k, v] of _searchParams) {
    if (k === 'v') {
      // v参数字段是由$符拼接的字符串
      const vals = v.split('$')
      queryParams[k] = vals.reduce((rst, item) => {
        const [key, value] = item.split('=')
        rst[key] = value
        return rst
      }, {})
    } else {
      queryParams[k] = v
    }
  }
  return queryParams
}
