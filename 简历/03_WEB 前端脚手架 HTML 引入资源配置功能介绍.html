    现在我们有了配置文件 html-config.cjs，有了资源信息 dataConfig，也有了 html 模版，现在就是要根据配置文件从资源信息里取出开启的资源替换到 html 模版中，最终生成一份完整的 html 文件。

    构建时配置 html 逻辑

    /**
 * @description 动态配置html
 * 
 * @param projectPath {string} 工程目录
 * @param htmlTpl {string} html模板字符串
 * @param htmlConfigData {Object} html配置数据 就是 html-config.cjs 中的配置
 * @param errorDeleteDir {boolean} 出错时是否删除工程
 * */
function setHtml(projectPath, htmlTpl, htmlConfigData, errorDeleteDir) {
  try {
    // ejs.render 的第二个参数，就是要替换 ejs 模版的数据对象
    const configData = {}

    // 从 html-config 中将 true 的脚本增加，false 的脚本删除
    for (const key in htmlConfigData) {
      const configVal = htmlConfigData[key]
      const js_key = `${key}Js`
      const css_key = `${key}Css`

      // 如果开启配置 则开始替换
      if (Object.is(configVal, true)) {
        // 替换 js + css
        if (key === 'mallCommonComponent') {
          configData[js_key] = formatTemplate(dataConfig[js_key], js_key)
          configData[css_key] = formatTemplate(dataConfig[css_key], css_key)
        } else {
          // 只替换 js
          configData[key] = formatTemplate(dataConfig[key], key)
        }
      } else if (Object.is(configVal, false)) {
        // 如果关闭配置 则要删除
        if (key === 'mallCommonComponent') {
          configData[js_key] = ''
          configData[css_key] = ''
        } else {
          configData[key] = ''
        }
      } else {
        // 带参数的
        // 烛龙监控
        if (key === 'dra') {
          configData[key] = ejs.render(formatTemplate(dataConfig[key], key), configVal)
        } else {
          configData[key] = ''
        }}
    }

    // 替换ejs模板
    htmlTpl = ejs.render(htmlTpl, configData)

    fs.writeFileSync(path.join(projectPath, './index.html'), pretty(htmlTpl, {
      ocd: true
    }))
  } catch (err) {
    console.log('设置html时出错', err)
    errorDeleteDir && fs.removeSync(projectPath)
    process.exit(0)
  }
}

    上面引用的 formatTemplate 函数代码

    // 给元素增加自定义属性
// <script src="..."></script> ==>> <script src="..." data-s="xx"></script>
// <link href="..." rel="stylesheet" /> ==>> <link href="..." rel="stylesheet" data-s="xx" />
function addElAttr (elStr, key) {
  let newStr = ''
  // 找到第一个 > 的位置
  const index = elStr.indexOf('>')
  // 判断前一个字符是不是 /
  if (elStr.charAt(index - 1) === '/') {
    newStr = `${elStr.substring(0, index - 1)} data-s="${key}" ${elStr.substring(index - 1)}`
  } else {
    newStr = `${elStr.substring(0, index)} data-s="${key}" ${elStr.substring(index)}`
  }
  return newStr
}

// 格式化脚本html片段
export function formatTemplate (arr, key) {
  // 给每一个资源增加自定义属性
  const newArr = arr.map(item => addElAttr(item, key))
  return newArr.join('\n')
}

    更新时配置 html 逻辑

    与构建项目时配置 html 逻辑不同的是，更新时要读取开发者项目中的 index.html 和 html-config.cjs 文件，此时要用 index.html 文件反向生成一份包含完整资源模版的 html 模版。

    这是更新的函数

    /**
 * 更新html
 * */
async function updateHtml () {
  try {
    const htmlConfig = require(path.join(process.cwd(), './html-config.cjs'))
    // 读取html字符串
    let htmlStr = fs.readFileSync(path.join(process.cwd(), './index.html'), 'utf-8')
    // 将html转换成模板
    htmlStr = replaceTpl(htmlStr, 'script')
    htmlStr = replaceTpl(htmlStr, 'link')
    // 去重
    for (const key in dataConfig) {
      htmlStr = duplicateRemove(htmlStr, `<%- ${key} %>`)
    }
    
    // 再执行 setHtml 逻辑
    setHtml(process.cwd(), htmlStr, htmlConfig, false)

    console.log(green('更新成功'))
  } catch (err) {
    console.log('更新html时出错', err)
    process.exit(0)
  }
}

    replaceTpl 函数代码

    // 将html转化成模板
export function replaceTpl (str, type) {
  if (!str) return str
  // 把 <script src="..." data-s="xx"></script> 替换成 <%- xx %>
  if (type === 'script') {
    return str.replace(/<script\b[^>]*data-s="([^"|']*)"[^>]*>([\s\S]*?)<\/script>/g, function () {
      return `<%- ${arguments[1]} %>`
    })
  }
  // 把 <link href="..." rel="stylesheet" data-s="xx" /> 替换成 <%- xx %>
  if (type === 'link') {
    return str.replace(/<link\b[^>]*data-s="([^"|']*)"[^>]*\/>/g, function () {
      return `<%- ${arguments[1]} %>`
    })
  }
  return str
}

    由于一个 key 对应的资源有可能是多个，所以替换成模版也有多个，所以要去重

    // 在html中标签所在位置的信息
export const posInfo = [
  '<\/title>',
  '<%- mallCommonComponentCss %>',
  '<%- base %>',
  '<%- risk %>',
  '<%- security %>',
  '<%- eventemiter %>',
  '<%- dra %>',
  '<%- mallCommonComponentJs %>',
]

// 获取数组索引
function getIndex (arr, str) {
  for (let i = 0, len = arr.length; i < len; i ++) {
    if (arr[i] === str) return i
  }
  return -1
}

/**
 * @description 链式插入算法
 *
 * @param htmlStr {string} html 模版
 * @param preEjsTpl {string} 前一个资源模版
 * @param insertEjsTplArr {string[]} 累计待插入资源模版
 * */
function insertLink (htmlStr, preEjsTpl, insertEjsTplArr) {
  const arr = htmlStr.split(preEjsTpl)
  // 如果 length 大于 1 说明有前一个模版，则走插入逻辑
  if (arr.length > 1) {
    const [first, ...rest] = arr
    return `${first}${preEjsTpl}${insertEjsTplArr.join('')}${rest.join('')}`
  } else {
    // 如果没有，将 preEjsTpl 保存到待插入资源数组中
    insertEjsTplArr.unshift(preEjsTpl)
    // 继续找前一个模版索引
    const preIndex = getIndex(posInfo, preEjsTpl) -1
    // 递归操作
    return insertLink(htmlStr, posInfo[preIndex], insertEjsTplArr)
  }
}

/**
 * @description 模板字符串去重
 * 
 * @param htmlStr {string} html 模版
 * @param ejsTpl {string} 去重的资源模版 例如：<%- base %>
 * */
export function duplicateRemove (htmlStr, ejsTpl) {
  const arr = htmlStr.split(ejsTpl)
  if (arr.length > 1) {
    const [first, ...rest] = arr
    return `${first}${ejsTpl}${rest.join('')}`
  } else {
    // htmlStr 中不含 ejsTpl
    // 前一个模板索引
    const preIndex = getIndex(posInfo, ejsTpl) -1
    // 插入前一个索引对应的模板
    return insertLink(htmlStr, posInfo[preIndex], [ejsTpl])
  }
}
